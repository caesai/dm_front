                            <Route path={'/gastronomy/order/:order_id'} element={<GastronomyOrderPage />} />
 
                            <Route path="*" element={<Navigate to="/" />} />
                        </Routes>
                    </Suspense>
                </Redirecter>
            )}
        </BrowserRouter>
    );
};
